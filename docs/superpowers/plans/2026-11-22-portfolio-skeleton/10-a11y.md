- [ ] **Step 2: Write the tests**

`tests/a11y/axe.spec.ts`

````ts
import AxeBuilder from '@axe-core/playwright';
import { expect, test } from '@playwright/test';
import { ALL_ROUTES } from '../e2e/routes';

for (const route of [...ALL_ROUTES, '/unknown-page/']) {
  test(`${route} has no serious accessibility violations`, async ({ page }) => {
    await page.goto(route);
    const { violations } = await new AxeBuilder({ page }).analyze();
    const serious = violations
      .filter((v) => v.impact === 'serious' || v.impact === 'critical')
      .map((v) => `${v.id}: ${v.nodes.map((n) => n.html).join(' ')}`);
    expect(serious).toEqual([]);
  });
}

test('the skip link moves focus to the main content', async ({ page }) => {
  await page.goto('/');
  await page.keyboard.press('Tab');
  await expect(page.getByRole('link', { name: 'Skip to content' })).toBeFocused();
  await page.keyboard.press('Enter');
  await expect(page).toHaveURL(/#main$/);
});
````

- [ ] **Step 4: Name the gate step (gate G8)**

`.github/actions/quality-gate/action.yml`

````yaml
name: Quality gate
description: Every automated check. Pull requests and deploys both run this.
inputs:
  github-token:
    description: Token for the GitHub GraphQL API. Empty means fixture data.
    default: ''
  site-url:
    description: Public URL of the site. Empty means a placeholder.
    default: ''
runs:
  using: composite
  steps:
    - uses: actions/setup-node@v7.1.0
      with:
        node-version-file: .nvmrc
        cache: npm
    - name: G1 install
      shell: bash
      run: npm ci
    - name: Audit report (does not block)
      shell: bash
      run: npm audit --audit-level=high || echo "::warning::npm audit found high severity issues"
    - name: G2 type check
      shell: bash
      run: npm run check
    - name: G3 and G4 unit tests and coverage
      shell: bash
      run: npm run test:unit
    - name: G5 content integrity and build
      shell: bash
      env:
        GH_GRAPHQL_TOKEN: ${{ inputs.github-token }}
        SITE_URL: ${{ inputs.site-url }}
      run: npm run build
    - name: Install Chromium
      shell: bash
      run: npx playwright install --with-deps chromium
    - name: G7 PDFs
      shell: bash
      run: npm run pdf
    - name: G6, G7 and G8 browser, PDF and accessibility tests
      shell: bash
      run: npm run test:e2e
    - name: Upload the test report
      if: failure()
      uses: actions/upload-artifact@v7.0.2
      with:
        name: playwright-report-${{ github.run_attempt }}
        path: playwright-report/
        if-no-files-found: ignore
````
